# Python packages for this repo's scripts. Install into a virtual environment
# that can still see KiCad's pcbnew module:
#
#   python3 -m venv --system-site-packages .venv     # Ubuntu: apt install python3-venv
#   .venv/bin/pip install -r requirements.txt
#
# With uv, name the system interpreter. KiCad's pcbnew is built for Ubuntu's
# Python 3.12, and a uv-managed 3.13 venv cannot import it at all:
#   uv venv --python /usr/bin/python3 --system-site-packages .venv
#   uv pip install -r requirements.txt
#
# Most scripts need nothing beyond the standard library
# (check_envelope.py, check_gates.py, sync_renders.py, software/transcribe.py).

# route_pcb.py (maze repair router) and export_metal_step.py.
# Ubuntu alternative: apt install python3-numpy (scripts/setup-ubuntu.sh does).
numpy>=1.24

# export_metal_step.py only: B-rep STEP of the 6061 tray for a CNC quote.
# Large download (OpenCascade); skip it if you are not quoting the metal tray.
cadquery>=2.4
